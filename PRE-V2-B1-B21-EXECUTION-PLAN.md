```bash
ensure_issue() {
  repo="$1" var="$2" title="$3" body="$4"; shift 4
  matches="$(gh issue list --repo "$repo" --state all --limit 200 --json number,title,url \
    | jq -r --arg title "$title" '.[] | select(.title == $title) | [.number,.url] | @tsv')"
  count="$(printf '%s\n' "$matches" | sed '/^$/d' | wc -l)"
  case "$count" in
    0) url="$(gh issue create --repo "$repo" --title "$title" --body-file "$body" "$@")"
       number="${url##*/}" ;;
    1) number="$(printf '%s\n' "$matches" | cut -f1)"
       url="$(printf '%s\n' "$matches" | cut -f2)" ;;
    *) echo "duplicate exact issue title: $title" >&2; return 4 ;;
  esac
  printf '%s=%q\n%s_URL=%q\n' "$var" "$number" "$var" "$url" >> .git/pre-v2-exec.env
}
```

Create body files under `.git/pre-v2-issue-bodies/`. Keep that directory private. After each call, source the env file and run `gh issue view` to prove the title and URL.
